import asyncio
import flet as ft
import flet_audio as fta  # pip install flet-audio


# ================= CONFIGURACIÓN =================
IMAGES = [
    "imagesp/parrot.png", "imagesp/dolphin.png", "imagesp/panda.png",
    "imagesp/turtles.png", "imagesp/hummingbird.png", "imagesp/rabbit.png",
    "imagesp/duck.png", "imagesp/eagle.png", "imagesp/cat.png",
    "imagesp/leopard.png",
]
CARD_BACK = "imagesp/card_back.png"

AUDIO_MAP = {
    "parrot": "audio/parrot.mp3",
    "dolphin": "audio/dolphin.mp3",
    "panda": "audio/panda.mp3",
    "turtles": "audio/turtles.mp3",
    "hummingbird": "audio/hummingbird.mp3",
    "rabbit": "audio/rabbit.mp3",
    "duck": "audio/duck.mp3",
    "eagle": "audio/eagle.mp3",
    "cat": "audio/cat.mp3",
    "leopard": "audio/leopard.mp3",
}


class MemoryGame:
    def __init__(self, page: ft.Page):
        self.page = page
        self.page.title = "El Rincón de Olga Lucia"
        self.page.bgcolor = ft.Colors.BLACK
        self.page.vertical_alignment = ft.MainAxisAlignment.CENTER
        self.page.horizontal_alignment = ft.CrossAxisAlignment.CENTER

        # Solo afecta escritorio (no web). Lo dejo por compatibilidad.
        self.page.window_width = 850
        self.page.window_height = 950

        # Responsivo: recalcular tamaños en móvil/rotación
        self.page.on_resized = self._on_page_resized

        self.grid: ft.GridView | None = None

        # juego
        self.difficulty = "Normal"
        self.cards: list[str] = []
        self.flipped_cards: list[int] = []
        self.matched_cards: set[int] = set()
        self.card_controls: list[ft.Container] = []
        self.player_score = 0
        self.robokko_score = 0
        self.current_turn = "Jugador"
        self.is_processing = False
        self.robokko_memory: dict[str, set[int]] = {}

        # AUDIO: mantener referencias vivas
        self.active_audios = []

        # UI
        self.score_text = ft.Text("", size=30, color=ft.Colors.YELLOW, weight=ft.FontWeight.BOLD)
        self.turn_text = ft.Text("", size=22, weight=ft.FontWeight.BOLD)

        self.show_main_menu()

    # ---------- RESPONSIVE LAYOUT ----------
    def _on_page_resized(self, e):
        self.apply_responsive_layout()

    def apply_responsive_layout(self):
        """
        Ajusta tamaño de cartas y grid según page.width/page.height.
        Mantiene 4 columnas x 5 filas.
        """
        if not self.grid or not self.card_controls:
            return

        w = self.page.width or 850
        h = self.page.height or 950

        cols = 4
        rows = 5

        side_padding = 16
        top_area_estimate = 190
        bottom_area_estimate = 70

        spacing = int(max(6, min(14, w * 0.02)))
        available_w = max(200, w - (side_padding * 2))
        available_h = max(200, h - top_area_estimate - bottom_area_estimate)

        card_size = int((available_w - spacing * (cols - 1)) / cols)

        # límites para que no se vea gigante ni minúsculo
        card_size = max(55, min(130, card_size))

        grid_h = card_size * rows + spacing * (rows - 1)
        if grid_h > available_h:
            scale = available_h / grid_h
            card_size = int(card_size * scale)
            card_size = max(45, card_size)

        grid_w = card_size * cols + spacing * (cols - 1)
        grid_h = card_size * rows + spacing * (rows - 1)

        self.grid.runs_count = cols
        self.grid.max_extent = card_size
        self.grid.spacing = spacing
        self.grid.run_spacing = spacing
        self.grid.width = grid_w
        self.grid.height = grid_h

        radius = max(10, int(card_size * 0.12))
        for c in self.card_controls:
            c.width = card_size
            c.height = card_size
            c.border_radius = radius
            if isinstance(c.content, ft.Image):
                c.content.fit = ft.BoxFit.CONTAIN

        self.page.update()

    # ---------- SALIR ----------
    def exit_app(self, _=None):
        """
        En escritorio: cierra la ventana/app.
        En web: no se puede cerrar la pestaña automáticamente -> muestra mensaje.
        """
        # Intentar cerrar ventana (desktop)
        try:
            self.page.window_close()
            return
        except Exception:
            pass

        # Web/mobile browser
        self.page.clean()
        self.page.overlay.clear()
        self.page.add(
            ft.Column(
                [
                    ft.Text("✅ Listo.", size=34, color=ft.Colors.YELLOW, weight=ft.FontWeight.BOLD),
                    ft.Text("En iPad / tablet, el navegador no permite cerrar la pestaña automáticamente.",
                            size=16, color=ft.Colors.WHITE),
                    ft.Text("Puedes cerrar esta pestaña ahora.", size=18, color=ft.Colors.WHITE),
                ],
                alignment=ft.MainAxisAlignment.CENTER,
                horizontal_alignment=ft.CrossAxisAlignment.CENTER,
            )
        )
        self.page.update()

    # ---------- AUDIO ----------
    def play_sound_from_image(self, image_path: str):
        filename = image_path.split("/")[-1]
        animal = filename.split(".")[0].lower()

        audio_src = AUDIO_MAP.get(animal)
        if not audio_src:
            return

        try:
            audio = fta.Audio(
                src=audio_src,
                autoplay=True,
                volume=1,
                balance=0,
                on_state_change=self._on_audio_state_changed,
            )

            # mantener vivo
            self.active_audios.append(audio)
            if len(self.active_audios) > 5:
                self.active_audios.pop(0)

        except Exception as e:
            print(f"Error creando audio para {animal}: {e}")

    def _on_audio_state_changed(self, e):
        try:
            pass
        except Exception as ex:
            print(f"Error en callback de audio: {ex}")

    # ---------- UI ----------
    def show_main_menu(self):
        self.page.clean()
        self.page.overlay.clear()
        self.active_audios.clear()

        title = ft.Text("El Rincón de Olga Lucia", size=50, color=ft.Colors.YELLOW, weight=ft.FontWeight.BOLD)

        btn_style = ft.ButtonStyle(
            color={
                ft.ControlState.DEFAULT: ft.Colors.BLACK,
                ft.ControlState.HOVERED: ft.Colors.BLACK,
            },
            bgcolor={
                ft.ControlState.DEFAULT: ft.Colors.YELLOW,
                ft.ControlState.HOVERED: ft.Colors.WHITE,
            },
            shape={ft.ControlState.DEFAULT: ft.RoundedRectangleBorder(radius=10)},
            animation_duration=300,
        )

        btn_facil = ft.Button("NIVEL 1 (FÁCIL)", width=300, height=50, style=btn_style,
                              on_click=lambda _: self.start_game("Fácil"))
        btn_medio = ft.Button("NIVEL 2 (MEDIO)", width=300, height=50, style=btn_style,
                              on_click=lambda _: self.start_game("Normal"))
        btn_dificil = ft.Button("NIVEL 3 (DIFÍCIL)", width=300, height=50, style=btn_style,
                                on_click=lambda _: self.start_game("Difícil"))

        # ✅ Botón circular SALIR (debajo de NIVEL 3)
        salir_btn = ft.Container(
            width=90,
            height=90,
            bgcolor=ft.Colors.YELLOW,
            border_radius=45,
            alignment=ft.Alignment(0, 0),  # <-- compatible con Flet 0.80.5
            ink=True,
            on_click=self.exit_app,
            content=ft.Text("SALIR", size=16, color=ft.Colors.BLACK, weight=ft.FontWeight.BOLD),
        )

        self.page.add(
            ft.Column(
                [
                    title,
                    ft.Text("Selecciona tu nivel de desafío:", size=20, color=ft.Colors.WHITE),
                    ft.Divider(height=20, color=ft.Colors.TRANSPARENT),
                    btn_facil,
                    ft.Divider(height=5, color=ft.Colors.TRANSPARENT),
                    btn_medio,
                    ft.Divider(height=5, color=ft.Colors.TRANSPARENT),
                    btn_dificil,
                    ft.Divider(height=18, color=ft.Colors.TRANSPARENT),
                    salir_btn,
                ],
                alignment=ft.MainAxisAlignment.CENTER,
                horizontal_alignment=ft.CrossAxisAlignment.CENTER
            )
        )
        self.page.update()

    def start_game(self, difficulty: str):
        self.difficulty = difficulty
        self.cards = IMAGES * 2
        random.shuffle(self.cards)

        self.flipped_cards.clear()
        self.matched_cards.clear()
        self.card_controls.clear()
        self.player_score = 0
        self.robokko_score = 0
        self.robokko_memory.clear()
        self.current_turn = "Jugador"
        self.is_processing = False
        self.active_audios.clear()

        self.page.clean()
        self.page.overlay.clear()

        # Grid responsivo (tamaños se recalculan)
        self.grid = ft.GridView(
            expand=False,
            runs_count=4,
            max_extent=110,
            spacing=10,
            run_spacing=10,
            width=500,
            height=500
        )

        for idx, img in enumerate(self.cards):
            card = ft.Container(
                content=ft.Image(src=CARD_BACK, fit=ft.BoxFit.CONTAIN),
                width=110,
                height=110,
                bgcolor=ft.Colors.GREY_900,
                border_radius=12,
                data={"idx": idx, "img": img},
                on_click=self.on_card_click_sync
            )
            self.card_controls.append(card)
            self.grid.controls.append(card)

        back_btn = ft.TextButton(
            "VOLVER AL MENÚ",
            on_click=lambda _: self.show_main_menu(),
            style=ft.ButtonStyle(color=ft.Colors.WHITE)
        )

        self.page.add(
            ft.Column(
                [
                    ft.Text(f"MODO: {self.difficulty.upper()}", size=15, color=ft.Colors.BLUE,
                            weight=ft.FontWeight.BOLD),
                    self.score_text,
                    self.turn_text,
                    ft.Row([self.grid], alignment=ft.MainAxisAlignment.CENTER),
                    back_btn
                ],
                alignment=ft.MainAxisAlignment.CENTER,
                horizontal_alignment=ft.CrossAxisAlignment.CENTER
            )
        )

        self.update_ui()
        self.apply_responsive_layout()

    # ---------- GAME ----------
    def on_card_click_sync(self, e: ft.ControlEvent):
        if self.is_processing or self.current_turn != "Jugador":
            return
        idx = e.control.data["idx"]
        self.reveal_card_sync(idx)

    def reveal_card_sync(self, idx: int):
        if idx in self.matched_cards or idx in self.flipped_cards:
            return

        card = self.card_controls[idx]
        img_path = card.data["img"]

        prob = {"Fácil": 0.3, "Normal": 0.6, "Difícil": 1.0}
        if random.random() < prob[self.difficulty]:
            self.robokko_memory.setdefault(img_path, set()).add(idx)

        card.content.src = img_path
        card.update()

        self.play_sound_from_image(img_path)

        self.flipped_cards.append(idx)

        if len(self.flipped_cards) == 2:
            self.is_processing = True
            self.page.run_task(self.check_match_delayed)

    async def check_match_delayed(self):
        if self.current_turn == "Robokko":
            await asyncio.sleep(2.0)

        await asyncio.sleep(1.2)
        await self.check_match()

    async def check_match(self):
        i1, i2 = self.flipped_cards

        if self.cards[i1] == self.cards[i2]:
            self.matched_cards.update({i1, i2})
            if self.current_turn == "Jugador":
                self.player_score += 1
            else:
                self.robokko_score += 1
            self.card_controls[i1].border = ft.Border.all(3, ft.Colors.GREEN)
            self.card_controls[i2].border = ft.Border.all(3, ft.Colors.GREEN)
            continuar = True
        else:
            self.card_controls[i1].content.src = CARD_BACK
            self.card_controls[i2].content.src = CARD_BACK
            continuar = False

        self.card_controls[i1].update()
        self.card_controls[i2].update()
        self.flipped_cards.clear()

        if not continuar:
            self.current_turn = "Robokko" if self.current_turn == "Jugador" else "Jugador"

        self.update_ui()
        self.is_processing = False

        if len(self.matched_cards) == len(self.cards):
            self.show_winner()
        elif self.current_turn == "Robokko":
            espera = {"Fácil": 1.5, "Normal": 1.0, "Difícil": 0.5}
            await asyncio.sleep(espera[self.difficulty])
            await self.robokko_turn()

    async def robokko_turn(self):
        t1, t2 = None, None
        for img, pos in self.robokko_memory.items():
            avail = [p for p in pos if p not in self.matched_cards]
            if len(avail) >= 2:
                t1, t2 = avail[0], avail[1]
                break

        if t1 is None:
            opts = [i for i in range(len(self.cards)) if i not in self.matched_cards]
            if len(opts) >= 2:
                t1 = random.choice(opts)
                opts.remove(t1)
                t2 = random.choice(opts)

        if t1 is not None and t2 is not None:
            self.reveal_card_sync(t1)
            await asyncio.sleep(1.0)
            self.reveal_card_sync(t2)

    def update_ui(self):
        self.score_text.value = f"JUGADOR: {self.player_score}   |   ROBOKKO: {self.robokko_score}"
        self.turn_text.value = f"TURNO: {self.current_turn.upper()}"
        self.turn_text.color = ft.Colors.GREEN if self.current_turn == "Jugador" else ft.Colors.RED
        self.page.update()

    def show_winner(self):
        if self.player_score > self.robokko_score:
            msg = "¡FELICIDADES, GANASTE!"
        elif self.player_score < self.robokko_score:
            msg = "GANÓ ROBOKKO"
        else:
            msg = "¡ES UN EMPATE!"

        def restart(_):
            self.page.dialog.open = False
            self.page.update()
            self.show_main_menu()

        dlg = ft.AlertDialog(
            modal=True,
            title=ft.Text(msg, weight=ft.FontWeight.BOLD, size=24),
            content=ft.Text(
                f"Marcador Final:\n\nJugador: {self.player_score}\nRobokko: {self.robokko_score}",
                size=18
            ),
            actions=[ft.Button(
                "JUGAR DE NUEVO",
                on_click=restart,
                style=ft.ButtonStyle(bgcolor=ft.Colors.YELLOW, color=ft.Colors.BLACK)
            )],
            actions_alignment=ft.MainAxisAlignment.CENTER,
        )

        self.page.dialog = dlg
        dlg.open = True
        self.page.update()


def main(page: ft.Page):
    MemoryGame(page)


import os

if __name__ == "__main__":
    port = int(os.environ.get("PORT", "8080"))
    ft.run(
        main,
        assets_dir="assets",
        host="0.0.0.0",
        port=port,
    )
